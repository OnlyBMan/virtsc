# Build Qemu fork

cd qemu-is1/
mkdir -p build && cd build
../configure --target-list=i386-softmmu \
             --enable-osmesa --disable-opengl \
             --enable-gtk --enable-sdl

# Create an overlay disk for easy undoing
chmod 444 $IMG
qemu-is1/build/qemu-img create -f qcow2 -F qcow2 -b $IMG ./work.qcow2

# Create the video & audio input FIFO
mkfifo ./is1-in-v ./is1-in-a

# Startup image
qemu/build/qemu-system-i386 \
  -name IS1 \
  -machine pc,acpi=off \
  -global i440FX.agp=on -global i440FX.agp-aperture-size=128M \
  -global piix3-ide.force-bus-master=on \
  -accel kvm -cpu pentium3 -m 512 -smp 1 \
  -drive file=./work.qcow2,format=qcow2,if=ide,cache=writeback -boot c \
  -vga cirrus \
  -netdev user,id=net0,net=10.100.102.0/24,host=10.100.102.1 \
  -device i82557b,netdev=net0 \
  -netdev user,id=net1,net=10.0.2.0/24,host=10.0.2.2,hostfwd=tcp:127.0.0.1:2222-10.0.2.15:22 \
  -device e1000-82545em,netdev=net1 \
  -rtc base=utc,clock=host \
  -serial file:./serial.log \
  -qmp unix:./qmp.sock,server,nowait \
  -device thunderstorm,id=tsc0,present=on,version=0x011a0012,\
input=bars,output=./is1-output,\
input-pipe=./is1-in-v,input-audio=./is1-in-a,\
stamp=host-ns,tstamp=host-s,timecode=utc,audio=silence \
  -device is1gl,id=is1gl0,mmio=0xfed10000,iobase=0x520 \
  -display gtk,zoom-to-fit=on,gl=off

# Interrupt boot and boot into `boot -s`

# Mount the drive
fsck -p
mount -u -o rw /
mount -a 

# Open firewall and bring up em0
/sbin/ipfw add 1 allow ip from any to any
ifconfig em0 inet 10.0.2.15 netmask 255.255.255.0

# Set password for root
echo password | pw usermod root -h 0

# Set static em0 IP
vi /etc/rc.conf
#   ifconfig_em0="inet 10.0.2.15 netmask 255.255.255.0"

# Copy the XF86Config-4.qemu-cirrus
fetch -o /etc/X11/XF86Config-4.qemu-cirrus \
      http://10.0.2.2:8000/XF86Config-4.qemu-cirrus
ln -sf XF86Config-4.qemu-cirrus /etc/X11/XF86Config-4

# Set device permissions for X
chmod 666 /dev/agpgart /dev/mem

# Enable debug mode 
vi /twc/util/rundomestic.sh

# Setup SSH access
mkdir -p /root/.ssh && chmod 700 /root/.ssh
fetch -o /root/.ssh/authorized_keys http://10.0.2.2:8000/authorized_keys
chmod 600 /root/.ssh/authorized_keys

# Remove root barred login
vi /etc/login.access
# Comment out the last line

# Continue loading
exit

# SSH into the is1 after it comes back up
ssh -p 2222 \
    -i ~/.ssh/is1_rsa \
    -o IdentitiesOnly=yes \
    -o KexAlgorithms=+diffie-hellman-group1-sha1,diffie-hellman-group-exchange-sha1 \
    -o HostKeyAlgorithms=+ssh-rsa,ssh-dss \
    -o PubkeyAcceptedAlgorithms=+ssh-rsa \
    -o Ciphers=+aes128-cbc,3des-cbc \
    -o MACs=+hmac-sha1 \
    -o StrictHostKeyChecking=no \
    -o UserKnownHostsFile=/dev/null \
    root@127.0.0.1

# Create folder for GL source (run in VM)
mkdir -p /usr/local/src

# Copy our source
scp -P 2222 \
    -i ~/.ssh/is1_rsa \
    -o IdentitiesOnly=yes \
    -o KexAlgorithms=+diffie-hellman-group1-sha1,diffie-hellman-group-exchange-sha1 \
    -o HostKeyAlgorithms=+ssh-rsa,ssh-dss \
    -o PubkeyAcceptedAlgorithms=+ssh-rsa \
    -o Ciphers=+aes128-cbc,3des-cbc \
    -o MACs=+hmac-sha1 \
    -o StrictHostKeyChecking=no \
    -o UserKnownHostsFile=/dev/null \
    ./tools/agp/libagpnv.c \
    ./tools/glfix/libglfix.c \
    ./tools/is1gl/is1gl.c \
    ./tools/is1gl/is1gl_ring.h \
    ./tools/is1gl/is1gl_ops.h \
    ./tools/is1gl/is1gl_gen_guest.h \
    root@127.0.0.1:/usr/local/src/

# Build GL (run in VM)
cd /usr/local/src
gcc -O2 -fPIC -shared -o /usr/local/lib/libagpnv.so libagpnv.c
gcc -O2 -fPIC -shared -I/usr/X11R6/include \
    -o /usr/local/lib/libglfix.so libglfix.c -L/usr/X11R6/lib -lGL
gcc -O2 -fPIC -shared -I. -I/usr/X11R6/include \
    -o /usr/local/lib/libis1gl.so is1gl.c -L/usr/X11R6/lib -lX11

# Install it as the system GL
mv /usr/X11R6/lib/libGL.so.1 /usr/X11R6/lib/libGL.so.1.mesa
cp /usr/local/lib/libis1gl.so /usr/X11R6/lib/libGL.so.1
ldconfig -m /usr/X11R6/lib

# Reboot the VM
reboot

# Send video/audio input
ffmpeg -hide_banner -loglevel warning -re -i ../vid.mkv \
  -map 0:v:0 \
  -vf "scale=720:480,vflip,format=yuyv422" \
  -r 30000/1001 -fps_mode cfr -f rawvideo -y ./is1-in-v \
  -map 0:a:0 -af aresample=48000 -ar 48000 -ac 2 -f s32le -y ./is1-in-a

# Enable NDI drain
make -C ./tools/is1ndi NDI_SDK="$HOME/NDI SDK for Linux"
LD_LIBRARY_PATH="$HOME/NDI SDK for Linux/lib/x86_64-linux-gnu" \
  ./tools/is1ndi/is1ndi --out ./is1-output --name IntelliStar
